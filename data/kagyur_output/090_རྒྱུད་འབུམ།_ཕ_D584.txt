པྲཛྙེ་ཤྲུ་ཏི་སྨྲྀ་ཏི་བི་ཛ་ཡེ། དྷཱིཿདྷཱ་ར་ཎཱི་ཡེ་སྭཱ་ཧཱ། འདི་བཟུང་བས་ཤེས་རབ་ཀྱི་ཕ་རོལ་ཏུ་ཕྱིན་པ་སྟོང་ཕྲག་བརྒྱ་པ་གཟུང་བར་འགྱུར་རོ། །ཤེས་རབ་ཀྱི་ཕ་རོལ་ཏུ་ཕྱིན་པ་སྟོང་ཕྲག་བརྒྱ་པ་གཟུང་བར་འགྱུར་བའི་གཟུངས་རྫོགས་སོ།། ༄༅། །དཀོན་མཆོག་གསུམ་
ལ་ཕྱག་འཚལ་ལོ། །ཨོཾ་ནྠཱཾ་ན་མཿསརྦ་བུད་དྷཱ་ནཱཾ། ཨོཾ་མཱཾ། བྷྲཱུཾ་མཱུཾ། ཏདྱ་ཐཱ། ཨོཾ་བྲཱུཾ། འདི་བཟུང་བས་འཕགས་པ་ཕལ་པོ་ཆེ་གཟུང་བར་འགྱུར་རོ། །འདྲི་བ་དང་། ཀློག་པ་ཡང་དེ་དང་འདྲའོ། །འཕགས་པ་ཕལ་པོ་ཆེ་གཟུང་བར་འགྱུར་བའི་གཟུངས་རྫོགས་སོ།། །།
